cmake_minimum_required(VERSION 3.16)
project(studio_media LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat>=61 libavcodec>=61 libavutil>=59 libswscale>=8)
if(NOT AV_libavformat_VERSION VERSION_LESS 62 OR NOT AV_libavcodec_VERSION VERSION_LESS 62)
  message(FATAL_ERROR "This development adapter is qualified against FFmpeg 7.x only; do not silently substitute another ABI")
endif()
find_package(nlohmann_json 3.11 REQUIRED)
add_library(studio_media SHARED src/studio_media.cpp)
target_include_directories(studio_media PUBLIC include)
target_compile_features(studio_media PRIVATE cxx_std_20)
target_compile_options(studio_media PRIVATE -Wall -Wextra -Werror)
target_link_libraries(studio_media PRIVATE PkgConfig::AV nlohmann_json::nlohmann_json)
include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(studio_media_abi_test tests/abi_test.cpp)
  target_link_libraries(studio_media_abi_test PRIVATE studio_media nlohmann_json::nlohmann_json Threads::Threads)
  foreach(case handles errors cancel)
    add_test(NAME media_${case} COMMAND studio_media_abi_test ${case})
    set_tests_properties(media_${case} PROPERTIES TIMEOUT 120)
  endforeach()
endif()
