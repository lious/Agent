```text

# 本机覆盖保存默认单文件.studioproj ZIP；目录包只新建/另建版本，不假定非空目录原子替换
 manifest.json                      # format/schema/minReader/project/snapshot hash
 snapshots/<docHash>.json           # canonical ProjectDocument，不存凭据
 originals/<sha256>                 # 用户选“嵌入”才复制；否则external locator引用
 assets/<sha256>                    # 生成片/图片/mask/音频等不可变资产
 proxies/<pipelineHash>/<sha256>    # 可重建，默认归档可排除

 sessions/                         # 仅用户勾选带会话归档时写脱敏事件/内容
 journal/commands.jsonl             # 可选审计导出；运行中真相在workspace SQLite
 checksums.json                     # 包内文件hash、size；不含自身hash
 recovery/latest-good.json          # 只指向验证完的snapshot
 extensions/                        # 未知字段/厂商数据保持，不执行代码

```

```text
-- workspace SQLite拟定核心DDL；所有API必须绑定ns，模型不能发SQL。
PRAGMA foreign_keys=ON;
-- ns是app/tenant/workspace的受信编码，不能由模型拼路径。
CREATE TABLE project_heads(ns TEXT NOT NULL,project_id TEXT NOT NULL,
 revision INTEGER NOT NULL CHECK(revision>=0),snapshot_hash TEXT NOT NULL,
 PRIMARY KEY(ns,project_id));
CREATE TABLE commands(ns TEXT NOT NULL,id TEXT NOT NULL,project_id TEXT NOT NULL,
 input_hash TEXT NOT NULL,expected_revision INTEGER NOT NULL CHECK(expected_revision>=0),
 before_ref TEXT NOT NULL,after_ref TEXT NOT NULL,inverse_ref TEXT NOT NULL,
 payload_json TEXT NOT NULL,actor_id TEXT NOT NULL,
 origin TEXT NOT NULL CHECK(origin IN ('user','agent')),
 PRIMARY KEY(ns,id),FOREIGN KEY(ns,project_id) REFERENCES project_heads(ns,project_id));
CREATE TABLE revisions(ns TEXT NOT NULL,project_id TEXT NOT NULL,
 revision INTEGER NOT NULL CHECK(revision>0),parent_revision INTEGER NOT NULL,
 snapshot_hash TEXT NOT NULL,command_id TEXT NOT NULL,
 PRIMARY KEY(ns,project_id,revision),UNIQUE(ns,command_id),
 FOREIGN KEY(ns,project_id) REFERENCES project_heads(ns,project_id),
 FOREIGN KEY(ns,command_id) REFERENCES commands(ns,id));
CREATE TABLE receipts(ns TEXT NOT NULL,receipt_key TEXT NOT NULL,
 input_hash TEXT NOT NULL,result_json TEXT NOT NULL,PRIMARY KEY(ns,receipt_key));
CREATE TABLE stream_heads(ns TEXT NOT NULL,stream_id TEXT NOT NULL,
 next_seq INTEGER NOT NULL CHECK(next_seq>0),PRIMARY KEY(ns,stream_id));
CREATE TABLE events(ns TEXT NOT NULL,stream_id TEXT NOT NULL,seq INTEGER NOT NULL CHECK(seq>0),
 event_id TEXT NOT NULL,type TEXT NOT NULL,payload_ref TEXT NOT NULL,created_at TEXT NOT NULL,
 PRIMARY KEY(ns,stream_id,seq),UNIQUE(ns,event_id),
 FOREIGN KEY(ns,stream_id) REFERENCES stream_heads(ns,stream_id));
CREATE TABLE outbox(ns TEXT NOT NULL,id TEXT NOT NULL,kind TEXT NOT NULL,
 input_hash TEXT NOT NULL,intent_ref TEXT NOT NULL,state TEXT NOT NULL,
 remote_id TEXT,attempt INTEGER NOT NULL CHECK(attempt>=0),
 lease_epoch INTEGER NOT NULL CHECK(lease_epoch>=0),PRIMARY KEY(ns,id));
CREATE TABLE blob_edges(ns TEXT NOT NULL,owner_kind TEXT NOT NULL,owner_id TEXT NOT NULL,
 owner_version TEXT NOT NULL,blob_hash TEXT NOT NULL,
 PRIMARY KEY(ns,owner_kind,owner_id,owner_version,blob_hash));
CREATE TABLE entity_versions(ns TEXT NOT NULL,project_id TEXT NOT NULL,
 entity_id TEXT NOT NULL,version INTEGER NOT NULL CHECK(version>=0),
 kind TEXT NOT NULL,content_hash TEXT NOT NULL,tombstoned INTEGER NOT NULL CHECK(tombstoned IN (0,1)),
 PRIMARY KEY(ns,project_id,entity_id,version),
 FOREIGN KEY(ns,project_id) REFERENCES project_heads(ns,project_id));
-- 重复entity/version同hash可复用；同版本异hash拒绝IMMUTABLE_VERSION_CONFLICT。
-- 初始revision0在project_heads登记；从revision1起commands/revisions一一对应。
-- 分配seq：同事务UPDATE stream_heads SET next_seq=next_seq+1 RETURNING next_seq-1；
-- 每stream连续递增，Branch继承用parent/fork引用，不复制父事件或假定分支seq连续。
-- 整数DB列内部有符号64位，JSON桥接统一十进制字符串；不经JS double。
-- 文档commit事务：UPDATE project_heads SET revision=:next,snapshot_hash=:hash
-- WHERE ns=:ns AND project_id=:pid AND revision=:expected；changes()!=1即回滚。
-- 同事务INSERT revisions/commands/events/receipts；blob pins在文档生效前已存在。
-- outbox只表示副作用意图，event replay不遍历执行outbox；专用dispatcher核验审批。
-- Session/Branch/Run/Proposal使用第9节字段建独立表，recordVersion CAS，
-- 不因聊天rename或轮询进度增加工程revision。FTS只索引允许保留的contentRef正文。
```

保存包与运行库分离：本地workspace的SQLite是事件/投影/receipt真相，原件及大媒体在CAS目录；.studioproj是可移出、可验证的快照。一次Project包可引用N个source blob，一个blob可被N个AssetVersion引用。外链工程体积小但不保证换机可播放；“收集并归档”嵌入所有选定原件/生成片，字体与会话需单独授权选择，模型只存引用。

下面是完整展开的Project JSON Schema 2020-12，不再是摘要。全部本地$defs及其required/properties/nullable/range已内嵌，无远程$ref；复制代码块即可得到schema文件。新必填字段和冻结版本引用属于破坏性变更，因此正式规格升级2.0.0，不能伪称1.x兼容新增。后面的命令Schema同样完整展开。结构校验不替代下文的跨引用、时间、权限和资源校验。
